if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "Configure from the repository root")
endif()

add_library(sandbox STATIC
    src/config.cpp
    src/apply_config.cpp
    src/apply_config.h
    src/process.cpp
    src/filesystem.cpp
    src/filesystem_access.h
    src/error_schema.h
    src/error/error.h
    src/error/error.cpp
    src/error/capture.cpp
    src/process/io_failure.h
    src/registry.cpp
)

target_include_directories(sandbox
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
)

target_compile_features(sandbox PUBLIC cxx_std_20)
target_link_libraries(sandbox
    PUBLIC nlohmann_json::nlohmann_json
)

if(WIN32)
    target_sources(sandbox PRIVATE
        src/window/process.cpp
        src/window/process.h
        src/window/appcontainer.cpp
        src/window/appcontainer.h
        src/window/job.cpp
        src/window/job.h
        src/window/raii.h
        src/window/filesystem.cpp
        src/window/filesystem.h
        src/window/acl.cpp
        src/window/acl.h
        src/window/identity.cpp
        src/window/identity.h
        src/window/state.cpp
        src/window/state.h
        src/window/error.h
    )
    target_link_libraries(sandbox PRIVATE
        Advapi32
        Bcrypt
        Userenv
    )
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(sandbox PRIVATE
        src/linux/process.cpp
        src/linux/process.h
        src/linux/landlock.cpp
        src/linux/landlock.h
        src/linux/filesystem.cpp
        src/linux/filesystem.h
        src/linux/identity.cpp
        src/linux/identity.h
        src/linux/state.cpp
        src/linux/state.h
    )
endif()

option(SANDBOX_BUILD_TESTS "Build sandbox tests" ON)

if(SANDBOX_BUILD_TESTS)
    enable_testing()

    add_executable(sandbox_error_schema_test tests/error_schema_test.cpp)
    target_link_libraries(sandbox_error_schema_test PRIVATE sandbox)
    target_include_directories(sandbox_error_schema_test PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
    )
    add_test(NAME sandbox_error_schema COMMAND sandbox_error_schema_test)
    add_executable(sandbox_registry_test
        tests/registry_test.cpp
    )
    target_link_libraries(sandbox_registry_test PRIVATE sandbox)
    target_include_directories(sandbox_registry_test PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
    )

    add_executable(sandbox_process_test
        tests/process_test.cpp
    )
    target_link_libraries(sandbox_process_test PRIVATE sandbox)
    target_include_directories(sandbox_process_test PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
    )

    add_test(NAME sandbox_registry COMMAND sandbox_registry_test)
    add_test(NAME sandbox_process COMMAND sandbox_process_test)
    set_tests_properties(sandbox_registry sandbox_process PROPERTIES RUN_SERIAL TRUE)
endif()
